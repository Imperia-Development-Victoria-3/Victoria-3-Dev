kurd_pacification.1 = {
	type = country_event
	placement = root
	title = kurd_pacification.1.t
	desc = kurd_pacification.1.desc
	flavor = kurd_pacification.1.f
	duration = 3
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	# Increase radicals of Kurds
	option = {
		name = kurd_pacification.1.a
		kurd_pacification_change_progress_bar = {
			CHANGE = -2
		}
		c:TUR = {
			add_radicals = {
				value = 0.05
				culture = cu:kurdish
			}
			trigger_event = {
				id = otto_kurd.1
			}
		}
	}
	# Sabotage Ottoman Kurdish state
	option = {
		name = kurd_pacification.1.b
		kurd_pacification_change_progress_bar = {
			CHANGE = -1
		}
		c:TUR = {
			random_scope_building = {
				limit = {
					state.state_region = {
						is_homeland = cu:kurdish
					}
				}
				add_modifier = {
					name = kurd_pacification_sabotage
					years = 1
					is_decaying = yes
				}
			}
			trigger_event = {
				id = otto_kurd.1
			}
		}
	}
	# Raid for resources
	option = {
		name = kurd_pacification.1.c
		kurd_pacification_change_progress_bar = {
			CHANGE = -1
		}
		c:TUR = {
			trigger_event = {
				id = otto_kurd.2
			}
		}
	}
}

# Centralize Tribes
# Meet with X prominent tribe, chance for success with corresponding modifiers.
# Assert Dominance over tribe, 60/40 chance for success, gives positive modifier if successful and negative modifier if not.
# Develop infrastructure in tribal holding, +10 Infrastructure State Modifier option for each state owned by Kurds, can’t do more than once. Per state.

kurd_pacification.2 = {
	type = country_event
	placement = root
	title = kurd_pacification.2.t
	desc = kurd_pacification.2.desc
	flavor = kurd_pacification.2.f
	duration = 3
	event_image = {
		video = "middleeast_islam"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	immediate = {
		random_scope_state = {
			limit = {
				NOT = {
					has_modifier = kurd_pacification_developed_state_infrastructure
				}
			}
			save_scope_as = random_state
		}
	}
	# Meet with X prominent tribe, chance for success with corresponding modifiers.
	option = {
		name = kurd_pacification.2.a
		random = {
			chance = 33
			# Add the modifier, whatever it is
			custom_tooltip = "positive modifier"
		}
	}
	# Assert Dominance over tribe, 60/40 chance for success, gives positive modifier if successful and negative modifier if not.
	option = {
		name = kurd_pacification.2.b
		random_list = {
			60 = {
				# Success!
				# Add positive modifier
				custom_tooltip = "positive modifier"
			}
			40 = {
				# Failure!
				# Add negative modifier
				custom_tooltip = "negative modifier"
			}
		}
	}
	# Develop infrastructure in tribal holding, +10 Infrastructure State Modifier option for each state owned by Kurds, can’t do more than once. Per state.
	option = {
		name = kurd_pacification.2.c
		trigger = {
			exists = scope:random_state
		}
		scope:random_state ?= {
			add_modifier = {
				name = kurd_pacification_developed_state_infrastructure
			}
		}
	}
}
