namespace = russification

russification.1 = {
	type = country_event

	title = russification.1.t
	desc = russification.1.d
	
	event_image = {
		video = "gfx/event_pictures/unspecific_military_parade.bk2"
	}
	option = {
		name = russification.1.a
		default_option = yes

		# Initialise the global variable
		set_global_variable = {
			name = russificating_finland
			value = 0
		}

		add_journal_entry = {
            type = russification_fi_je
        }
        c:FIN = {
            add_journal_entry = {
                type = russification_fi_je
            }
			hidden_effect = { # To avoid tooltips..
				trigger_event = {
					id = russification.2 # Set the global variable to 50, done here so the JE doesnt get confused.
				}
			}
        }
	}
	
	option = {
		name = russification.1.b

		remove_global_variable = russification_taken
	}
}

# A hidden event, to properly set the global variable to 50 without the JE getting confused.
russification.2 = {
	type = country_event
	hidden = yes
	immediate = {
		change_global_variable = {
			name = russificating_finland
			add = 50
		}
	}
}
