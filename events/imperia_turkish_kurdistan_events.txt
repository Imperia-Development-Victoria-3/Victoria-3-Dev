# [Police] Crack down on Kurdish Insurgents, +2 Governor, higher negative modifiers, more radicals, but less turmoil effects.

otto_kurd.1 = {
	type = country_event
	placement = root
	title = otto_kurd.1.t

	event_image = {
		video = "unspecific_factory_closed"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	# Decimate Kurdish tribe, +1 Revolt, Negative modifiers for Kurds, prevents agitation for an additional year.
	option = {
		name = otto_kurd.1.a
		default_option = yes
		kurd_pacification_change_progress_bar = {
			CHANGE = 1
		}
		c:KUR = {
			add_modifier = {
				name = kurd_pacification_decimated_kurdish_tribe
				years = 2
				is_decaying = yes
			}
			kill_population_percent = {
				value = 0.01
				culture = cu:kurdish
			}
			add_radicals = {
				value = 0.05
				culture = cu:kurdish
			}
			post_notification = kurd_pacification_decimated_kurdish_tribe
			custom_tooltip = {
				text = kurd_pacification_button_cooldown_extended_tt
				remove_variable ?= kurd_pacification_button_cooldown
				set_variable = {
					name = kurd_pacification_button_cooldown
					year = 2
				}
			}
		}
		ai_chance = {
			base = 20
		}
	}
	# Propaganda campaign, +1 Governor, Negative modifiers for Ottomans, but gives Kurdish loyalists
	option = {
		name = otto_kurd.1.b
		kurd_pacification_change_progress_bar = {
			CHANGE = -1
