		add_radicals = {

			value = 0.2
			pop_type = aristocrats
		}

	}
	option = { #Fund reformist papers
		name = imperia_hundred_days.1.b
		scope:journal_entry = {
			add_progress = {
				value = 10
				name = hundred_days_reform_enthusiasm_progress_bar
			}
		}
		add_modifier = {
			name = modifier_paying_journalist
			months = 60
		}
	}
	option = { #Ignore them
		name = imperia_hundred_days.1.c
		scope:journal_entry = {
			add_progress = {
				value = -5
				name = hundred_days_reform_enthusiasm_progress_bar
			}

			value = 0.05
		}
