namespace = imperia_dual_monarchy
# Choose what kind of Hungary you (HUN) want to send as ultimatum for the 1st Hungarian Revolution
imperia_dual_monarchy.1 = {
	type = country_event
	title = imperia_dual_monarchy.1.t
	desc = imperia_dual_monarchy.1.desc
	flavor = imperia_dual_monarchy.1.f
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 3
	immediate = {
		overlord = {
			save_scope_as = overlord_scope
		}
	}
	# Greater Hungary
	option = {
		name = imperia_dual_monarchy.1.a
		default_option = yes
		custom_label = if_they_accept_tt
		show_as_tooltip = {
			make_independent = yes
		}
		overlord = {
			trigger_event = {
				id = imperia_dual_monarchy.2
				popup = yes
			}
		}
		set_variable = {
			name = hungary_sent_greater_hungary_ultimatum_to_overlord
			value = yes
		}
		ai_chance = {
			base = 100
		}
	}
	# Lesser Hungary
	option = {
		name = imperia_dual_monarchy.1.b
		custom_label = if_they_accept_tt
		show_as_tooltip = {
			every_scope_state = {
				limit = {
					state_region = {
						NOR = {
							this = s:STATE_TRANSDANUBIA
							this = s:STATE_CENTRAL_HUNGARY
							this = s:STATE_BEKES
							this = s:STATE_WEST_SLOVAKIA
							this = s:STATE_EAST_SLOVAKIA
						}
					}
				}
				set_state_owner = scope:overlord_scope
			}
			make_independent = yes
		}
		overlord = {
			trigger_event = {
				id = imperia_dual_monarchy.2
				popup = yes
			}
		}
		set_variable = {
			name = hungary_sent_greater_hungary_ultimatum_to_overlord
			value = no
		}
		ai_chance = {
			base = 0
		}
	}
	option = {
		name = imperia_dual_monarchy.1.c
		remove_global_variable = hungary_sent_ultimatum_to_overlord
		ai_chance = {
			base = 0
		}
	}
}

# Austria gets to answer the ultimatum
imperia_dual_monarchy.2 = {
	type = country_event
	title = imperia_dual_monarchy.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = imperia_dual_monarchy.2.desc
				trigger = {
					c:HUN.var:hungary_sent_greater_hungary_ultimatum_to_overlord = yes
				}
			}
			triggered_desc = {
				desc = imperia_dual_monarchy.2.desc2
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = imperia_dual_monarchy.2.f
	event_image = {
		video = "unspecific_ruler_speaking_to_people"
	}
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	# 1 Month to choose
	duration = 1
	immediate = {
		# Done like this because root might be Hungary, if I remember event-chaining correctly..?
		this = {
			save_scope_as = this_scope
		}
	}
	# Accept greater Hungary
	option = {
		name = imperia_dual_monarchy.2.a
		trigger = {
			c:HUN.var:hungary_sent_greater_hungary_ultimatum_to_overlord = yes
		}
		c:HUN = {
			make_independent = yes
		}
		# ~1% chance of accepting the greater ultimatum
		ai_chance = {
			base = 1
		}
	}
	# Accept lesser Hungary
	option = {
		name = imperia_dual_monarchy.2.b
		trigger = {
			c:HUN.var:hungary_sent_greater_hungary_ultimatum_to_overlord = no
		}
		c:HUN = {
			every_scope_state = {
				limit = {
					state_region = {
						# Keep these states
						NOR = {
							this = s:STATE_TRANSDANUBIA
							this = s:STATE_CENTRAL_HUNGARY
							this = s:STATE_BEKES
							this = s:STATE_WEST_SLOVAKIA
							this = s:STATE_EAST_SLOVAKIA
						}
					}
				}
				set_state_owner = scope:this_scope
			}
			make_independent = yes
		}
		# ~2% chance of accepting the lesser ultimatum
		ai_chance = {
			base = 2
		}
	}
	# Decline the ultimatum
	option = {
		name = imperia_dual_monarchy.2.c
		default_option = yes
		c:HUN = {
			create_diplomatic_play = {
				name = hungarian_revolution
				target_country = scope:this_scope
				type = dp_imperia_hungarian_revolt
				add_war_goal = {
					holder = scope:this_scope
					target_country = c:HUN
					type = make_dominion
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
	after = {
		# After the ultimatum, no matter what, transfer the characters loyal to Austria back home
		# The governor and his son, and any commanders not of Hungarian culture
		c:HUN = {
			every_scope_character = {
				limit = {
					OR = {
						has_template = imperia_joseph_habsburg_template
						has_template = imperia_stephen_habsburg_template
						AND = {
							commander_rank > 0
							NOT = {
								has_culture = cu:hungarian
							}
						}
					}
				}
				transfer_character = c:AUS
			}
			# Activate something resembling the 12 points of the Hungarian Revolution
			activate_law = law_type:law_parliamentary_republic
			activate_law = law_type:law_universal_suffrage
			activate_law = law_type:law_elected_bureaucrats
			activate_law = law_type:law_total_separation
			activate_law = law_type:law_national_guard
			activate_law = law_type:law_interventionism
			activate_law = law_type:law_proportional_taxation
			activate_law = law_type:law_homesteading
			activate_law = law_type:law_protected_speech
			# Set Lajos Kossuth as its president
			if = {
				limit = {
					any_scope_character = {
						has_template = agitator_lajos_kossuth
					}
				}
				random_scope_character = {
					limit = {
						has_template = agitator_lajos_kossuth
					}
					set_character_as_ruler = yes
				}
			}
		}
	}
}

# Austrian emperor Ferdinand I abdicates in favour of Franz Joseph
imperia_dual_monarchy.3 = {
	type = country_event
	title = imperia_dual_monarchy.3.t
	desc = imperia_dual_monarchy.3.desc
	flavor = imperia_dual_monarchy.3.f
	gui_window = event_window_2char
	left_icon = scope:ruler_scope
	right_icon = scope:heir_scope
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"
	# 2 Month to choose
	duration = 2
	trigger = {
		ruler = {
			has_template = aus_ferdinand_I_habsburg_template
		}
	}
	immediate = {
		ruler = {
			save_scope_as = ruler_scope
		}
		heir = {
			save_scope_as = heir_scope
			set_character_as_ruler = yes
		}
	}
	option = {
		name = imperia_dual_monarchy.3.a
		default_option = yes
		custom_tooltip = following_has_happened_tt
		show_as_tooltip = {
			scope:heir_scope = {
				set_character_as_ruler = yes
			}
			if = {
				limit = {
					c:HUN.overlord = c:AUS
				}
				add_journal_entry = {
					type = je_imperia_austrian_neo_absolutism
				}
				c:HUN = {
					add_journal_entry = {
						type = je_imperia_passive_resistance
					}
				}
			}
		}
		ai_chance = {
			base = 100
		}
	}
}

# Great powers, European majors and Austrian neighbours can send aid in the conflict
imperia_dual_monarchy.4 = {
	type = country_event
	title = imperia_dual_monarchy.4.t
	desc = imperia_dual_monarchy.4.desc
	flavor = imperia_dual_monarchy.4.f
	event_image = {
		video = "unspecific_politicians_arguing"
	}
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	duration = 2
	immediate = {
		c:HUN = {
			save_scope_as = hungary_scope
		}
		c:AUS = {
			save_scope_as = austria_scope
		}
	}
	# Help Austria with a unique effect
	option = {
		name = imperia_dual_monarchy.4.a
		trigger = {
			imperia_hungarian_revolution_country_has_unique_aid_effects = yes
		}
		c:AUS = {
			# This automagically checks which country is sending aid, and gives the appropriate effect
			imperia_hungarian_revolution_foreign_aid_effect = yes
		}
		ai_chance = {
			base = 75
			modifier = {
				# Ottomans, France and Britain gets bad effects from helping Austria, so is less likely to do so
				trigger = {
					OR = {
						this = c:TUR
						this = c:FRA
						this = c:GBR
					}
				}
				add = -50
			}
		}
	}
	# Helps Hungary with a unique effect
	option = {
		name = imperia_dual_monarchy.4.b
		trigger = {
			imperia_hungarian_revolution_country_has_unique_aid_effects = yes
		}
		c:HUN = {
			# This automagically checks which country is sending aid, and gives the appropriate effect
			imperia_hungarian_revolution_foreign_aid_effect = yes
		}
		ai_chance = {
			base = 25
			modifier = {
				# France gets good effects from helping Hungary, so is more likely to do so
				trigger = {
					this = c:FRA
				}
				add = 50
			}
			modifier = {
				# Ottomans gets significantly bad effects from helping Hungary, so is less likely to do so
				trigger = {
					this = c:TUR
				}
				add = -10
			}
			modifier = {
				# Spain, Netherlands and Two Sicilies gets mixed effects from helping Hungary, so is a bit more likely to do so
				trigger = {
					OR = {
						this = c:SPA
						this = c:NET
						this = c:SIC
					}
				}
				add = 10
			}
		}
	}
	# Help Austria with a generic effect
	option = {
		name = imperia_dual_monarchy.4.c
		trigger = {
			imperia_hungarian_revolution_country_has_unique_aid_effects = no
		}
		c:AUS = {
			add_modifier = {
				name = hungarian_revolution_generic_aid_modifier
			}
		}
		ai_chance = {
			base = 25
		}
	}
	# Help Hungary with a generic effect
	option = {
		name = imperia_dual_monarchy.4.d
		trigger = {
			imperia_hungarian_revolution_country_has_unique_aid_effects = no
		}
		c:HUN = {
			add_modifier = {
				name = hungarian_revolution_generic_aid_modifier
			}
		}
		ai_chance = {
			base = 25
		}
	}
	# No help
	option = {
		name = imperia_dual_monarchy.4.e
		# If you're one of the unique aid countries, you will usually get an effect from not helping
		if = {
			limit = {
				imperia_hungarian_revolution_country_has_unique_aid_effects = yes
			}
			imperia_hungarian_revolution_foreign_aid_effect = yes
		}
		ai_chance = {
			base = 50
			modifier = {
				# Prussia and Ottomans gets minor bad effects from not helping, so is less likely to do so
				trigger = {
					OR = {
						this = c:PRU
						this = c:TUR
					}
				}
				add = -10
			}
			modifier = {
				# Russia gets bad effects from not helping, so is less likely to do so
				trigger = {
					this = c:RUS
				}
				add = -25
			}
			modifier = {
				# France gets mixed effect from not helping
				trigger = {
					this = c:FRA
				}
				add = 10
			}
		}
	}
}
