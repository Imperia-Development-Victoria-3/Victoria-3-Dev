imperia_passive_resistance_suppression_progress_bar = {
	name = "imperia_passive_resistance_suppression_progress_bar"
	desc = "imperia_passive_resistance_suppression_progress_bar_desc"
	default_bad = yes
	weekly_progress = {
		if = {
			limit = {
				exists = c:HUN
				exists = c:AUS
			}
			add = {
				desc = "passive_resistance_suppression_progress_from_economic_dependence_tt"
				value = global_var:hun_economic_dependence_on_aus
				divide = 4
			}
			add = {
				desc = "passive_resistance_suppression_progress_from_sol_tt"
				if = {
					limit = {
						c:HUN.average_sol > 12
					}
					value = -0.1
				}
				if = {
					limit = {
						c:HUN.average_sol < 12
					}
					value = 0.05
				}
			}
		}
	}
	start_value = 50
	min_value = 0
	max_value = 500
}

imperia_passive_resistance_deadlock_progress_bar = {
	name = "imperia_passive_resistance_deadlock_progress_bar"
	desc = "imperia_passive_resistance_deadlock_progress_bar_desc"
	default = yes
	weekly_progress = {
		if = {
			limit = {
				exists = c:HUN
				exists = c:AUS
			}
			# Deadlock when the values are within 10 points of each other
			if = {
				limit = {
					global_var:passive_resistance_deadlock_difference < 10
					global_var:passive_resistance_deadlock_difference > -10
				}
				add = {
					desc = "passive_resistance_deadlock_progress_from_deadlock_tt"
					value = 0.48
				}
			}
			# Lead when the values are more than 20 points apart
			else_if = {
				limit = {
					OR = {
						global_var:passive_resistance_deadlock_difference > 20
						global_var:passive_resistance_deadlock_difference < -20
					}
				}
				add = {
					desc = "passive_resistance_deadlock_progress_from_lead_tt"
					value = -0.96
				}
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 100
}
