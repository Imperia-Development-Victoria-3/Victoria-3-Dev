je_kurd_governorate_institute_ottomanism_progress_bar = {
	name = "je_kurd_governorate_institute_ottomanism_progress_bar"
	desc = "je_kurd_governorate_institute_ottomanism_progress_bar_desc"
	default_green = yes
	weekly_progress = {
		# Advance progress bar by 1 if there are no insurrectionary movements
		if = {
			limit = {
				any_political_movement = {
					count = all
					is_insurrectionary = no
				}
			}
			add = {
				desc = "no_insurrectionary_movements"
				value = 1
			}
		}
		###
		# Do a reset to 0 if there are insurrectionary movements present (or a civil war)
		# Done this way because honestly I'm not entirely sure how the civil war triggers work
		if = {
			limit = {
				OR = {
					any_political_movement = {
						is_insurrectionary = yes
					}
					any_civil_war = {
						count > 0
					}
				}
			}
			add = {
				desc = "insurrectionary_movement_present"
				value = -52
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 52
}

je_kurd_governorate_crush_emirs_progress_bar = {
	name = "je_kurd_governorate_crush_emirs_progress_bar"
	desc = "je_kurd_governorate_crush_emirs_progress_bar_desc"
	default_green = yes
	weekly_progress = {
		if = {
			limit = {
				any_civil_war = {
					count = 0
				}
			}
			add = {
				desc = "internal_peace"
				value = 1
			}
		}
		if = {
			limit = {
				any_civil_war = {
					count > 0
				}
			}
			add = {
				desc = "civil_war_present"
				value = -365
			}
		}
	}
	start_value = 0
	min_value = 0
	max_value = 365
}
