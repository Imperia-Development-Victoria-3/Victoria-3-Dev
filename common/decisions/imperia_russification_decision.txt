russification_finland = {
	is_shown = {
		exists = c:FIN
		this = c:RUS


		c:FIN = {
  			is_subject_of = root
		}

		NOT = { has_global_variable = russification_taken }

		NOT = {
			ROOT = {
				is_at_war = yes
			}
		}
	}

	when_taken = {
		trigger_event = {
			id = russification.1
			popup = yes
		} 
		
		set_global_variable = russification_taken
	}
}
