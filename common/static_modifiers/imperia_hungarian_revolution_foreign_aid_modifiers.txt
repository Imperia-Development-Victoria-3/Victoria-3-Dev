## Foreign Aid Modifiers for Great Powers, European majors and Austrian neighbours
# Prussia, Russia, Ottomans, France, Great Britain, Spain, Netherlands and Sicily gets unique effects
####################################################################################################
# First off is the generic aid modifier, available to countries if they're not one of the above
# This gets applied to Austria and Hungary equally
hungarian_revolution_generic_aid_modifier = {
    icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_positive.dds
}
## Prussian modifiers
# For Austria (good effects)
hungarian_revolution_good_prussian_aid_modifier = {
    icon = gfx/interface/icons/timed_modifier_icons/modifier_flag_positive.dds
    interest_group_ig_landowners_approval_add = 3
    interest_group_ig_intelligentsia_approval_add = -1
    country_radicalism_increases_full_acceptance_mult = -0.2
    country_political_strength_full_acceptance_mult = 0.15
    state_turmoil_effects_mult = 0.1
    country_expenses_add = 6000
}
# For Hungary (bad effects)
hungarian_revolution_bad_prussian_aid_modifier = {
    icon = gfx/interface/icons/timed_modifier_icons/modifier_flag_negative.dds
    interest_group_ig_landowners_approval_add = -1
    interest_group_ig_intelligentsia_approval_add = 2
    country_radicalism_increases_full_acceptance_mult = 0.25
    country_political_strength_full_acceptance_mult = -0.10
    state_turmoil_effects_mult = 0.15
    country_expenses_add = 6000
}
# For Prussia, if they stayed out of the conflict (minorly bad effects)
hungarian_revolution_no_prussian_aid_modifier = {
    icon = gfx/interface/icons/timed_modifier_icons/modifier_flag_negative.dds
    state_turmoil_effects_mult = 0.1
}
