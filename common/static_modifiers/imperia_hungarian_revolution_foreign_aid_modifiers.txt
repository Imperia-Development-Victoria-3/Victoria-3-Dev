# Prussia, Russia, Ottomans, France, Great Britain, Spain, Netherlands and Sicily gets unique effects

# First off is the generic aid modifier, available to countries if they're not one of the above
# This gets applied to Austria and Hungary equally

    icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_positive.dds
}

# For chosen war participant

    icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_positive.dds
}

    icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_positive.dds
}

    icon = gfx/interface/icons/timed_modifier_icons/modifier_rifle_negative.dds
