imperia_passive_resistance_hungarian_resistance_won_effect = {
	c:HUN = {
		create_diplomatic_play = {
			name = hungarian_revolution
			target_country = c:AUS
			type = dp_imperia_hungarian_revolt
			add_war_goal = {
				holder = scope:this_scope
				target_country = c:HUN
				type = annex_country
			}
		}
		if = {
			limit = {
				var:passive_resistance_underground_network_strength > 1
			}
			hidden_effect = {
				remove_modifier ?= passive_resistance_underground_network
			}
			# The strength is increased per button press after the addition of the modifier, so we subtract 1 here
			add_modifier = {
				name = passive_resistance_underground_network
				multiplier = {
					value = c:HUN.var:passive_resistance_underground_network_strength
					subtract = 1
				}
				is_decaying = yes
				years = 2
			}
		}
		remove_variable ?= passive_resistance_underground_network_strength
		remove_variable ?= hungarian_art_type
		remove_global_variable ?= passive_resistance_deadlock_difference
	}
}

			name = imperia_passive_resistance_state_modifier
		}
	}
	# Boosts the suppression and liberty desire progress bars PER CONFISCATED STATE!!
	# Currently set to 10 suppression and 8 (10x0.8) liberty desire

		add_progress = {
			value = imperia_passive_resistance_punishment_suppression_per_confiscated_state
			name = imperia_passive_resistance_suppression_progress_bar
		}

			value = imperia_passive_resistance_punishment_hungarian_resistance_per_confiscated_state
			name = imperia_passive_resistance_hungarian_resistance_progress_bar
		}

		add_progress = {
			value = imperia_passive_resistance_punishment_suppression_per_confiscated_state

imperia_hungarian_revolution_foreign_aid_effect = {
	## Used in the scope of the recipient, with root being the sender
	# If sender is Prussia..
	if = {
		limit = {
			root = c:PRU
		}
		imperia_hungarian_revolution_prussia_aid_effect = yes
	}
	# Russia
	else_if = {
		limit = {
			root = c:RUS
		}
		imperia_hungarian_revolution_russia_aid_effect = yes
	}
	# Ottomans
	else_if = {
		limit = {
			root = c:TUR
		}
		imperia_hungarian_revolution_ottoman_aid_effect = yes
	}
	# France
	else_if = {
		limit = {
			root = c:FRA
		}
		imperia_hungarian_revolution_france_aid_effect = yes
	}
	# Britain
	else_if = {
		limit = {
			root = c:GBR
		}
		imperia_hungarian_revolution_britain_aid_effect = yes
	}
	# Spain
	else_if = {
		limit = {
			root = c:SPA
		}
		imperia_hungarian_revolution_spain_aid_effect = yes
	}
	# Netherlands
	else_if = {
		limit = {
			root = c:NET
		}
		imperia_hungarian_revolution_netherlands_aid_effect = yes
	}
	# Two Sicilies
	else_if = {
		limit = {

		}
		imperia_hungarian_revolution_two_sicilies_aid_effect = yes
	}
}

imperia_hungarian_revolution_prussia_aid_effect = {
	# this = Austria / Hungary
	# root = Prussia
	if = {
		limit = {
			this = c:AUS
		}
		# Add effects to Austria here
		add_modifier = {
			name = hungarian_revolution_prussian_aid_modifier
		}
		root = {
			add_modifier = {
				name = hungarian_revolution_prussian_aid_to_austria_modifier
			}
		}
	}
	else_if = {
		limit = {
			this = c:HUN
		}
		# Add effects to Hungary here
		add_modifier = {
			name = hungarian_revolution_prussian_aid_modifier
		}
		root = {
			add_modifier = {
				name = hungarian_revolution_prussian_aid_to_hungary_modifier
			}
		}
	}
	else = {
		# Else assume Prussia stayed out of the conflict
		add_modifier = {
			name = hungarian_revolution_no_prussian_aid_modifier
		}
	}
}
