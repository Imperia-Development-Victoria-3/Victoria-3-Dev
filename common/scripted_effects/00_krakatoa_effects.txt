krakatoa_tsunami_effect = {
	every_scope_state = {
		limit = {
			OR = {
				region = sr:region_indochina
				region = sr:region_indonesia
				region = sr:region_south_china
			}
			is_coastal = yes
			state_region = {
				NOT = {
					has_variable = krakatoa_tsunami_hit
				}
			}
		}
		state_region = {
			set_variable = krakatoa_tsunami_hit
			add_devastation = 30
			start_harvest_condition = flood
		}
	}
	every_scope_state = {
		limit = {
			OR = {
				region = sr:region_zanj
				region = sr:region_ethiopia
				region = sr:region_southern_africa
				region = sr:region_madras
				region = sr:region_bengal
				region = sr:region_bombay
				region = sr:region_japan
				region = sr:region_oceania
			}
			is_coastal = yes
			state_region = {
				NOT = {
					has_variable = krakatoa_tsunami_hit
				}
			}
		}
		state_region = {
			set_variable = krakatoa_tsunami_hit
			add_devastation = 10
			start_harvest_condition = flood
		}
	}
}
