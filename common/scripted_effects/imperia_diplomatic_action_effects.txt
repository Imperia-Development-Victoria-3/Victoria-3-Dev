# GUNS TREATY
imperia_set_guns_treaty = {
	if = {
		limit = {
			scope:actor = {
				has_technology_researched = gunsmithing
			}
		}
		scope:target_country = {
			set_variable = imperia_gunsmithing_treaty
		}
	}
	if = {
		limit = {
			scope:actor = {
				has_technology_researched = rifling
			}
		}
		scope:target_country = {
			set_variable = imperia_rifling_treaty
		}
	}
	if = {
		limit = {
			scope:actor = {
				has_technology_researched = repeaters
			}
		}
		scope:target_country = {
			set_variable = imperia_repeaters_treaty
		}
	}
	if = {
		limit = {
			scope:actor = {
				has_technology_researched = bolt_action_rifles
			}
		}
		scope:target_country = {
			set_variable = imperia_bolt_action_rifles_treaty
		}
	}
}

imperia_reset_guns_treaty = {
	remove_variable ?= imperia_gunsmithing_treaty
	remove_variable ?= imperia_rifling_treaty
	remove_variable ?= imperia_repeaters_treaty
	remove_variable ?= imperia_bolt_action_rifles_treaty
}

# ADDING WAR GOALS
imperia_add_war_goal_effect = {
	random_diplomatic_play = {
		limit = {
			is_war = yes
			any_scope_play_involved = {
				this = scope:target_country
			}
			any_scope_play_involved = {
				this = root
			}
		}
		
		if = {
			limit = {
				initiator = root
			}
			every_scope_initiator_ally = {
				imperia_compensate_war_support = yes
				add_modifier = {
					name = imperia_adding_wargoal_defense
					years = 2
				}
			}
		}
		else = {
			every_scope_initiator_ally = {
				imperia_compensate_war_support = yes
				add_modifier = {
					name = imperia_adding_wargoal_defense
					years = 2
				}			
			}
		}

		switch = {
			trigger = $WARGOAL$
			conquer_state = {
				add_war_goal = {
					holder = root
					type = conquer_state
					target_state = scope:second_state
					primary_demand = no
					target_country = scope:target_country
				}
			}
			return_state = {
				add_war_goal = {
					holder = root
					type = return_state
					target_state = scope:second_state
					primary_demand = no
					target_country = scope:target_country
				}
			}
			take_treaty_port = {
				add_war_goal = {
					holder = root
					type = take_treaty_port
					target_state = scope:second_state
					primary_demand = no
					target_country = scope:target_country
				}
			}
			war_reparations = {
				add_war_goal = {
					holder = root
					type = war_reparations
					primary_demand = no
					target_country = scope:target_country
				}
			}
			ban_slavery = {
				add_war_goal = {
					holder = root
					type = ban_slavery
					primary_demand = no
					target_country = scope:target_country
				}
			}
			regime_change = {
				add_war_goal = {
					holder = root
					type = regime_change
					primary_demand = no
					target_country = scope:target_country
				}
			}
			contain_threat = {
				add_war_goal = {
					holder = root
					type = contain_threat
					primary_demand = no
					target_country = scope:target_country
				}
			}
			foreign_investment_rights = {
				add_war_goal = {
					holder = root
					type = foreign_investment_rights
					primary_demand = no
					target_country = scope:target_country
				}
			}
			humiliation = {
				add_war_goal = {
					holder = root
					type = humiliation
					primary_demand = no
					target_country = scope:target_country
				}
			}
			make_dominion = {
				add_war_goal = {
					holder = root
					type = make_dominion
					primary_demand = no
					target_country = scope:target_country
				}
			}
			make_protectorate = {
				add_war_goal = {
					holder = root
					type = make_protectorate
					primary_demand = no
					target_country = scope:target_country
				}
			}
			transfer_subject = {
				add_war_goal = {
					holder = root
					type = transfer_subject
					primary_demand = no
					target_country = scope:target_country
				}
			}
			liberate_subject = {
				add_war_goal = {
					holder = root
					type = liberate_subject
					primary_demand = no
					target_country = scope:target_country
				}
			}
		}
	}
}

imperia_compensate_war_support = {
	prev.war = {
		if = {
			limit = {
				has_war_support = {
					target = prev
					value < -90
				}
			}
			add_war_war_support = {
				target = prev
				value = 110
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < -70
				}
			}
			add_war_war_support = {
				target = prev
				value = 90
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < -70
				}
			}
			add_war_war_support = {
				target = prev
				value = 90
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < -50
				}
			}
			add_war_war_support = {
				target = prev
				value = 70
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < -30
				}
			}
			add_war_war_support = {
				target = prev
				value = 50
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < -10
				}
			}
			add_war_war_support = {
				target = prev
				value = 30
			}
		}
		else_if = {
			limit = {
				has_war_support = {
					target = prev
					value < 0
				}
			}
			add_war_war_support = {
				target = prev
				value = 20
			}
		}
	}
}
