	texture = "gfx/interface/icons/production_method_icons/picks_and_shovels.dds"

	# Most harmful form of coal, pollution wise. 30 - 40% more polluting

	state_modifiers = {
		# Normally no pollution this early, give 5 pollution

		}
	}
}
