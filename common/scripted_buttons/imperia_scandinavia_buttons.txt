je_scandinavian_union_vote_decentralization_button = {
    name = "scandinavian_union_vote_decentralization_button"
    desc = "scandinavian_union_vote_decentralization_button_desc"
    
    visible = {
        always = yes
    }
    
    possible = {
        NOT = {
            has_variable = je_scandinavian_union_voting_decentralization
        }
    }
    
	selected = {
		has_variable = je_scandinavian_union_voting_decentralization
	}

    effect = {
        custom_tooltip = {
            text = je_scandinavian_union_vote_decentralization_button_tt
            set_variable = je_scandinavian_union_voting_decentralization
            remove_variable ?= je_scandinavian_union_voting_nothing
            remove_variable ?= je_scandinavian_union_voting_centralization
        }
    }

    ai_chance = {
        value = 0
        if = {
            limit = {
                is_subject = yes
                overlord ?= {
                    has_variable = je_scandinavian_union_voting_decentralization
                }
            }
            add = 100
        }
        else_if = {
            limit = {
                is_subject = no
            }
            add = 5
        }
    }
}
