je_kurd_pacification = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	group = je_group_historical_content
	scripted_progress_bar = kurd_pacification_core_progress_bar
	scripted_button = kurd_pacification_agitate_ottomans_button
	scripted_button = kurd_pacification_centralize_tribes_button
	scripted_button = kurd_pacification_ottoman_cooperation_button
	status_desc = {
		first_valid = {
			triggered_desc = {
				# Below 13
				trigger = {
					scope:journal_entry = {
						scripted_bar_progress = {
							name = kurd_pacification_core_progress_bar
							value < 13
						}
					}
				}
				desc = je_kurd_pacification_status_governorship
			}
			triggered_desc = {
				# Above 13
				trigger = {
					scope:journal_entry = {
						scripted_bar_progress = {
							name = kurd_pacification_core_progress_bar
							value > 13
						}
					}
				}
				desc = je_kurd_pacification_status_revolt
			}
			triggered_desc = {
				# Exact 13
				trigger = {
					scope:journal_entry = {
						scripted_bar_progress = {
							name = kurd_pacification_core_progress_bar
							value = 13
						}
					}
				}
				desc = je_kurd_pacification_status_balanced
			}
		}
	}
	immediate = {
		# For localization purposes
		ig:ig_armed_forces = {
			save_scope_as = armed_forces_ig
		}
		ig:ig_landowners = {
			save_scope_as = landowners_ig
		}
	}
	complete = {
		custom_tooltip = {
			text = je_kurd_pacification_progress_bar_full
			scope:journal_entry = {
				scripted_bar_progress = {
					name = kurd_pacification_core_progress_bar
					value = 25
				}
			}
		}
	}
	fail = {
		custom_tooltip = {
			text = je_kurd_pacification_progress_bar_empty
			scope:journal_entry = {
				scripted_bar_progress = {
					name = kurd_pacification_core_progress_bar
					value = 0
				}
			}
		}
	}
	on_fail = {
		add_journal_entry = {
			type = je_kurd_governorate_main
		}
	}
	invalid = {
		is_subject = no
	}
	should_be_pinned_by_default = yes
}

je_kurd_governorate_main = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content
	immediate = {
		add_journal_entry = {
			type = je_kurd_governorate_institute_ottomanism
		}
		add_journal_entry = {
			type = je_kurd_governorate_new_jannisaries
		}
		add_journal_entry = {
			type = je_kurd_governorate_crush_the_emirs
		}
	}
	complete = {
		NOR = {
			has_journal_entry = je_kurd_governorate_institute_ottomanism
			has_journal_entry = je_kurd_governorate_new_jannisaries
			has_journal_entry = je_kurd_governorate_crush_the_emirs
		}
	}
	invalid = {
		is_subject = no
	}
	on_complete = {
		add_modifier = {
			name = kurd_loyal_population
		}
	}
}

		}

	}
}
