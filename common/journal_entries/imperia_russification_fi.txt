russification_fi_je = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_internal_affairs

	is_shown_when_inactive = {
		OR = {
  			c:FIN ?= root
  			c:RUS ?= root
		}
	}
	possible = {
		has_global_variable = russification_taken
	}
	progressbar = yes

	immediate = {
		if = {
			limit = {
				NOT = {
					has_global_variable = russificating_finland
				}
			}
			set_global_variable = {
				name = russificating_finland
				value = 50
			}
		}
	}

	current_value = {
		value = global_var:russificating_finland
	}

	goal_add_value = {
		value = 50
	}

	invalid = {
		ROOT = {
			is_at_war = yes
		}
	}

	on_invalid = {
		remove_global_variable = russification_taken
		remove_global_variable = russificating_finland
	}

	complete = {
		custom_tooltip = {
			text = russification_complete
			global_var:russificating_finland >= 100
		}
	}

	on_complete = {
		c:RUS = { annex = c:FIN }
	}

	fail = {
		custom_tooltip = {
			text = russification_failed
			global_var:russificating_finland <= 0
		}
	}

	on_fail = {
		c:FIN = { make_independent = ROOT}
		remove_global_variable = russification_taken
	}


	should_be_pinned_by_default = yes
	can_deactivate = yes
}
