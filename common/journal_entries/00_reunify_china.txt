je_reunify_china = {
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/invention_icons/wargaming.dds"
	
	group = je_group_foreign_affairs
	
	modifiers_while_active = {
		modifier_chinese_warlord_era
	}
	
	is_shown_when_inactive = {
		OR = {
			country_has_primary_culture = cu:han
			country_has_primary_culture = cu:manchu
		}
		OR = {
			has_global_variable = china_shatters
			AND = {
				has_variable = china_warlord_explosion
				var:china_warlord_explosion <= 3
			}
		}
	}
	
	possible = {
		always = no
	}
	
	complete = {
		trigger_if = {
			limit = {
				global_var:chinese_central_government ?= this
			}
			custom_tooltip = {
				text = no_independent_chinese_countries_tt
				any_country = {
					filter = {
						OR = {
							any_primary_culture = {
								has_discrimination_trait = heritage_han
							}
							country_has_primary_culture = cu:yi
							country_has_primary_culture = cu:manchu
						}
						OR = {
							is_in_geographic_region = geographic_region_china
							is_in_geographic_region = geographic_region_manchuria_old
							is_in_geographic_region = geographic_region_central_asia
							is_in_geographic_region = geographic_region_himalayas
						}
					}
					OR = {
						THIS = ROOT
						is_subject_of = ROOT
					}
					percent = 1
				}
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					global_var:chinese_central_government ?= {
						is_country_alive = yes
					}
				}
				custom_tooltip = {
					text = central_government_defeated_tt
					OR = {
						NOT = {
							exists = global_var:chinese_central_government
						}
						global_var:chinese_central_government ?= {
							is_country_alive = no
						}
					}
				}
			}
			custom_tooltip = {
				text = no_independent_chinese_countries_tt
				any_country = {
					filter = {
						OR = {
							any_primary_culture = {
								has_discrimination_trait = heritage_han
							}
							country_has_primary_culture = cu:yi
							country_has_primary_culture = cu:manchu
						}
						OR = {
							is_in_geographic_region = geographic_region_china
							is_in_geographic_region = geographic_region_manchuria_old
							is_in_geographic_region = geographic_region_central_asia
							is_in_geographic_region = geographic_region_himalayas
						}
					}
					OR = {
						THIS = ROOT
						is_subject_of = ROOT
					}
					percent = 1
				}
			}
		}
		OR = {
			owns_entire_state_region = STATE_NANJING
			owns_entire_state_region = STATE_BEIJING
		}
		trigger_if = {
			limit = {
				NOT = { c:TPG ?= this }
			}
			custom_tooltip = {
				text = has_formed_china_tt
				c:CHI ?= this
			}
		}
		NOT = {
			has_law_or_variant = law_type:law_peasant_levies
		}
	}
	
	on_complete = {
		remove_global_variable = chinese_central_government
	}
	
	weight = 1000
}
