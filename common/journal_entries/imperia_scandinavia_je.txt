je_scandinavian_nationalism = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_scandinavian_unification
	# Make a global_variable progress bar
	# Done as a global so the Nordics can share the progress
	current_value = {
		value = global_var:scandinavian_nationalism_progress
	}
	goal_add_value = {
		# The progress bar should be between 0 and 1000 to allow for more granularity
		add = 1000
	}
	progressbar = yes
	immediate = {
		# This section pops immediately when the JE starts
		# Initialize the progress bar, hopefully just once
		# I'll admit, the if-limit is mostly there so latecomers wont reset the progress bar :skull_emoji:
		if = {
			limit = {
				NOT = {
					has_global_variable = scandinavian_nationalism_progress
				}
			}
			set_global_variable = {
				name = scandinavian_nationalism_progress
				value = 0
			}
			set_global_variable = {
				name = scandinavian_nationalism_progress_change_since_last_week
				value = 0
			}
			set_global_variable = {
				name = scandinavian_nationalism_failure_timer
				value = 0
			}
		}
	}
	is_shown_when_inactive = {
		# Check if any primary culture is part of the Scandinavian culture group
		# Instead of checking each specific culture
		any_primary_culture = {
			has_discrimination_trait = heritage_nordic
		}
		# The '?=' means that if an overlord exist, they must also have a primary culture in the Scandinavian culture group
		# If it's just a '=' there MUST be an overlord
		overlord ?= {
			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}
	}
	possible = {
		# Check that every country with a Scandinavian primary culture have nationalism researched
		any_country = {
			filter = {
				any_primary_culture = {
					has_discrimination_trait = heritage_nordic
				}
				is_subject = no
			}
			count = all
			has_technology_researched = nationalism
		}
		# Check that every independent Scandinavian country has at least cordial relations with each other independent Scandinavian country
		# Creates a nonsensical tooltip so a custom tooltip is needed
		# Uses a n*2 check, which is not great but should be fine given the low number of countries involved (2-3, depending on released nations)
		custom_tooltip = {
			text = scandinavian_nationalism_inter_scandi_relations_good_tt
			any_country = {
				filter = {
					any_primary_culture = {
						has_discrimination_trait = heritage_nordic
					}
					is_subject = no
				}
				count = all
				any_country = {
					filter = {
						any_primary_culture = {
							has_discrimination_trait = heritage_nordic
						}
						is_subject = no
						this != prev
					}
					count = all
					relations:prev >= relations_threshold:cordial
				}
			}
		}
	}
	complete = {
		custom_tooltip = {
			text = scandinavian_nationalism_progress_at_100p_tt
			has_global_variable = scandinavian_nationalism_progress
			global_var:scandinavian_nationalism_progress >= 1000
		}
	}
	fail = {
		# 5 years of negative progress
		custom_tooltip = {
			text = scandinavian_nationalism_failure_tt
			has_global_variable = scandinavian_nationalism_failure_timer
			global_var:scandinavian_nationalism_failure_timer >= 260	#5*52 weeks
		}
	}
	on_fail = {
		if = {
			limit = {
				has_modifier = scandinavian_currency_union
			}
			remove_modifier = scandinavian_currency_union
		}
		if = {
			limit = {
				has_modifier = scandinavian_freedom_of_movement_act
			}
			remove_modifier = scandinavian_freedom_of_movement_act
		}
	}
	on_monthly_pulse = {
		effect = {
			# Add missing modifiers for latecomers
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_modifier = scandinavian_currency_union
					}
					NOT = {
						has_modifier = scandinavian_currency_union
					}
				}
				add_modifier = {
					name = scandinavian_currency_union
				}
			}
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_modifier = scandinavian_freedom_of_movement_act
					}
					NOT = {
						has_modifier = scandinavian_freedom_of_movement_act
					}
				}
				add_modifier = {
					name = scandinavian_freedom_of_movement_act
				}
			}
			# Create mutual defense pacts for latecomers (and newly freed Scandis)
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_variable = scandinavian_mutual_defense_act_completed
					}
					NOT = {
						has_variable = scandinavian_mutual_defense_act_completed
					}
					is_subject = no
				}
				set_variable = scandinavian_mutual_defense_act_completed
				every_country = { # Create defensive pacts between all relevant Scandinavian countries
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						is_subject = no
						this != root
						NOT = { # Avoid double treaties
							any_scope_treaty = {
								binds = root
								any_scope_article = {
									has_type = defensive_pact
								}
							}
						}
					}
					save_scope_as = second_country_scope
					create_treaty = {
						name = treaty_name_scandinavian_mutual_defense_act
						first_country = root
						second_country = scope:second_country_scope
						is_draft = no
						binding_period = { years = 100 }
						articles_to_create = {
							{
								article = defensive_pact
							}
						}
					}
				}
			}
		}
	}
	on_weekly_pulse = {
		effect = {
			# For the tooltip, find the difference between current progress and previous weeks progress
			save_scope_value_as = {
				name = old_scandinavian_nationalism_progress
				value = global_var:scandinavian_nationalism_progress
			}
			## WEEKLY CALCULATIONS FOR SCANDINAVIAN NATIONALISM PROGRESS BAR ##
			# If the current country has okay relations with another Scandi country...
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root >= relations_threshold:cordial
					}
				}
				change_global_variable = {
					name = scandinavian_nationalism_progress
					add = {
						# Add to the progress bar, depending on how positive the relations are
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:cordial
							}
							add = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:amicable
							}
							add = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:friendly
							}
							add = scandinavian_nationalism_progress_tick_small
						}
					}
				}
			}
			# And if the current country has poor relations with another Scandi country...
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root <= relations_threshold:poor
					}
				}
				change_global_variable = {
					name = scandinavian_nationalism_progress
					add = {
						# This is gonna be a negative given the below conditions, so its fine to use add here
						# Subtract from the progress bar, depending on how negative the relations are
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:poor
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:cold
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:hostile
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
					}
				}
			}
			# If at war with fellow Scandinavian
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_war_with = root
					}
				}
				every_country = {
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						has_war_with = root
					}
					change_global_variable = {
						name = scandinavian_nationalism_progress
						subtract = scandinavian_nationalism_progress_tick_large
					}
				}
			}
			# If in a war against non-Scandi country, with an independent Scandi as your ally
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_ally_of = root
					}
					any_country = {
						count = 0
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_enemy_of = root
					}
				}
				# This is a bit wonky.. If there's a diplo play against a non-Scandi featuring 2 Scandis, and an inter-Scandi war going on it'll cancel each other out
				every_country = {
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_ally_of = root
					}
					change_global_variable = {
						name = scandinavian_nationalism_progress
						add = scandinavian_nationalism_progress_tick_medium
					}
				}
			}
			# If the mutual defense act is broken (independent country leaving the defensive pact)
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_variable = scandinavian_mutual_defense_act_completed
						is_subject = no
						this != root
						NOT = {
							any_scope_treaty = {
								binds = root
								any_scope_article = {
									has_type = defensive_pact
								}
							}
						}
					}
				}
				every_country = {
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						has_variable = scandinavian_mutual_defense_act_completed
						is_subject = no
						this != root
						NOT = {
							any_scope_treaty = {
								binds = root
								any_scope_article = {
									has_type = defensive_pact
								}
							}
						}
					}
					change_global_variable = {
						name = scandinavian_nationalism_progress
						subtract = scandinavian_nationalism_progress_tick_large
					}
				}
			}
			## END OF WEEKLY CALCULATIONS
			# Clamp that variable so it stays between 0 and 1000
			clamp_global_variable = {
				name = scandinavian_nationalism_progress
				min = 0
				max = {
					value = 250
					# Any country completing the Currency Union JE increases the max by 250
					if = {
						limit = {
							any_country = {
								has_journal_entry = je_scandinavian_nationalism
								has_modifier = scandinavian_currency_union
							}
						}
						add = 250
					}
					# Any country completing the Freedom of Movement Act JE increases the max by 250
					if = {
						limit = {
							any_country = {
								has_journal_entry = je_scandinavian_nationalism
								has_modifier = scandinavian_freedom_of_movement_act
							}
						}
						add = 250
					}
				}
			}
			# For the tooltip, find the difference between current progress and previous weeks progress
			# Only done once per week to save on performance and to avoid a certain stutter that appears otherwise
			if = {
				limit = {
					any_country = {
						count = 0
						has_journal_entry = je_scandinavian_nationalism
						country_rank > root.country_rank
					}
				}
				set_global_variable = {
					name = scandinavian_nationalism_progress_change_since_last_week
					value = {
						value = global_var:scandinavian_nationalism_progress
						subtract = scope:old_scandinavian_nationalism_progress
					}
				}
				# If the progress is going back, increase the failure timer
				if = {
					limit = {
						global_var:scandinavian_nationalism_progress_change_since_last_week < 0
					}
					change_global_variable = {
						name = scandinavian_nationalism_failure_timer
						add = 1 # Add one week
					}
				}
				else = {
					# Otherwise, reset the failure timer
					set_global_variable = {
						name = scandinavian_nationalism_failure_timer
						value = 0
					}
				}
			}
		}
	}
	status_desc = {
		# Special status description for when the Scandinavian Nationalism JE is in danger of failing
		triggered_desc = {
			desc = je_scandinavian_nationalism_danger_of_failure
			trigger = {
				has_global_variable = scandinavian_nationalism_failure_timer
				global_var:scandinavian_nationalism_failure_timer > 0
			}
		}
		# A special status description for the Scandinavian Nationalism progress bar, telling how much it progressed last week
		triggered_desc = {
			desc = status_scandinavian_nationalism
			trigger = {
				has_global_variable = scandinavian_nationalism_progress
			}
		}
		# Tells you if its increasing from positive relations
		triggered_desc = {
			desc = je_scandinavian_nationalism_positive_relations
			trigger = {
				any_country = {
					has_journal_entry = je_scandinavian_nationalism
					relations:root >= relations_threshold:cordial
				}
			}
		}
		# Tells you if its increasing from negative relations
		triggered_desc = {
			desc = je_scandinavian_nationalism_negative_relations
			trigger = {
				any_country = {
					has_journal_entry = je_scandinavian_nationalism
					relations:root <= relations_threshold:poor
				}
			}
		}
		# Tells you if its decreasing from infighting
		triggered_desc = {
			desc = je_scandinavian_nationalism_infighting
			trigger = {
				any_country = {
					has_journal_entry = je_scandinavian_nationalism
					has_war_with = root
				}
			}
		}
		# Tells you if its increasing from fighting outsiders
		triggered_desc = {
			desc = je_scandinavian_nationalism_fighting_against_outsiders
			trigger = {
				any_country = {
					has_journal_entry = je_scandinavian_nationalism
					is_diplomatic_play_ally_of = root
				}
				any_country = {
					count = 0
					has_journal_entry = je_scandinavian_nationalism
					is_diplomatic_play_enemy_of = root
				}
			}
		}
		# Tells you if its decreasing from broken mutual defense act
		triggered_desc = {
			desc = je_scandinavian_nationalism_broken_mutual_defense_act
			trigger = {
				any_country = {
					has_journal_entry = je_scandinavian_nationalism
					has_variable = scandinavian_mutual_defense_act_completed
					is_subject = no
					this != root
					NOT = {
						any_scope_treaty = {
							binds = root
							any_scope_article = {
								has_type = defensive_pact
							}
						}
					}
				}
			}
		}
		# Tells you the current capacity bonus based on completed JEs
		desc = je_scandinavian_nationalism_current_cap_is
		first_valid = {
			triggered_desc = {
				desc = onehundred_percent
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_variable = scandinavian_mutual_defense_act_completed
					}
				}
			}
			triggered_desc = {
				desc = seventyfive_percent
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_modifier = scandinavian_freedom_of_movement_act
					}
				}
			}
			triggered_desc = {
				desc = fifty_percent
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_modifier = scandinavian_currency_union
					}
				}
			}
			desc = twentyfive_percent
		}
	}
	weight = 100
	should_be_pinned_by_default = yes
}

	group = je_group_scandinavian_unification
	is_shown_when_inactive = {
		any_primary_culture = {
			has_discrimination_trait = heritage_nordic

		overlord ?= {
			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}
	}
	possible = {
		any_primary_culture = {
			has_discrimination_trait = heritage_nordic
		}
		overlord ?= {
			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}
		has_journal_entry = je_scandinavian_nationalism
		custom_tooltip = {
			text = scandinavian_nationalism_progress_at_25p_tt
			# Remember this is out of 1000
			has_global_variable = scandinavian_nationalism_progress
			global_var:scandinavian_nationalism_progress >= 250
		}
	}
	complete = {
		any_country = {
			filter = {
				has_journal_entry = je_scandinavian_currency_union

			count = all

			has_technology_researched = pan-nationalism
		}
		# This part must be active for 5 years before it can conclude
		je:je_scandinavian_currency_union = {
			journal_entry_age >= 1825	#5*365
		}
	}
	on_complete = {
		add_modifier = {
			name = scandinavian_currency_union
		}
	}
}

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_scandinavian_unification
	possible = {
		any_primary_culture = {
			has_discrimination_trait = heritage_nordic
		}
		overlord ?= {
			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}
		has_journal_entry = je_scandinavian_nationalism
		any_country = {
			has_journal_entry = je_scandinavian_nationalism
			has_modifier = scandinavian_freedom_of_movement_act
		}
		custom_tooltip = {
			text = scandinavian_nationalism_progress_at_75p_tt
			# Remember this is out of 1000
			has_global_variable = scandinavian_nationalism_progress
			global_var:scandinavian_nationalism_progress >= 750
		}
	}
	complete = {
		OR = {
			AND = {
				any_country = { # 1 (one) relevant Scandi country must be independent
					filter = {
						has_journal_entry = je_scandinavian_nationalism
					}
					count = 1
					is_subject = no
				}
				any_country = { # And ALL relevant Scandi SUBJECTS must have a Scandi overlord 
					filter = {
						has_journal_entry = je_scandinavian_nationalism
						is_subject = yes
					}
					count = all
					overlord = {
						any_primary_culture = {
							has_discrimination_trait = heritage_nordic
						}
					}
				}
			}
			any_country = { # Or all relevant Scandi countries have multilateral alliances researched
				filter = {
					has_journal_entry = je_scandinavian_nationalism
				}
				count = all
				has_technology_researched = multilateral_alliances
			}
		}
	}
	on_complete = {
		if = { # Only for independent countries
			limit = {
				is_subject = no
			}
			set_variable = scandinavian_mutual_defense_act_completed
			every_country = { # Create defensive pacts between all relevant Scandinavian countries
				limit = {
					has_journal_entry = je_scandinavian_nationalism
					is_subject = no
					this != root
					NOT = { # Avoid double treaties
						any_scope_treaty = {
							binds = root
							any_scope_article = {
								has_type = defensive_pact
							}
						}
					}
				}
				save_scope_as = second_country_scope
				create_treaty = {
					name = treaty_name_scandinavian_mutual_defense_act
					first_country = root
					second_country = scope:second_country_scope
					is_draft = no
					binding_period = { years = 100 }
					articles_to_create = {
						{
							article = defensive_pact
						}
					}
				}
			}
		}
	}
}
