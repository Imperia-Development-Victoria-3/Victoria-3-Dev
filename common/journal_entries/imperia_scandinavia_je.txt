je_scandinavian_nationalism = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_scandinavian_unification
	# Make a global_variable progress bar
	# Done as a global so the Nordics can share the progress
	current_value = {
		value = global_var:scandinavian_nationalism_progress
	}
	goal_add_value = {
		# The progress bar should be between 0 and 1000 to allow for more granularity
		add = 1000
	}
	progressbar = yes
	immediate = {
		# This section pops immediately when the JE starts
		# Initialize the progress bar, hopefully just once
		# I'll admit, the if-limit is mostly there so latecomers wont reset the progress bar :skull_emoji:
		if = {
			limit = {
				NOT = {
					has_global_variable = scandinavian_nationalism_progress
				}
			}
			set_global_variable = {
				name = scandinavian_nationalism_progress
				value = 0
			}
			set_global_variable = {
				name = scandinavian_nationalism_progress_change_since_last_week
				value = 0
			}
		}
	}
	is_shown_when_inactive = {
		# Check if any primary culture is part of the Scandinavian culture group
		# Instead of checking each specific culture
		any_primary_culture = {
			has_discrimination_trait = heritage_nordic
		}
		# The '?=' means that if an overlord exist, they must also have a primary culture in the Scandinavian culture group
		# If it's just a '=' there MUST be an overlord
		overlord ?= {
			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}
	}
	possible = {
		# Check that every country with a Scandinavian primary culture have nationalism researched
		any_country = {
			filter = {
				any_primary_culture = {
					has_discrimination_trait = heritage_nordic
				}
				is_subject = no
			}
			count = all
			has_technology_researched = nationalism
		}
		# Check that every independent Scandinavian country has at least cordial relations with each other independent Scandinavian country
		# Creates a nonsensical tooltip so a custom tooltip is needed
		# Uses a n*2 check, which is not great but should be fine given the low number of countries involved (2-3, depending on released nations)
		custom_tooltip = {
			text = scandinavian_nationalism_inter_scandi_relations_good_tt
			any_country = {
				filter = {
					any_primary_culture = {
						has_discrimination_trait = heritage_nordic
					}
					is_subject = no
				}
				count = all
				any_country = {
					filter = {
						any_primary_culture = {
							has_discrimination_trait = heritage_nordic
						}
						is_subject = no
						this != prev
					}
					count = all
					relations:prev >= relations_threshold:cordial
				}
			}
		}
	}
	complete = {
		custom_tooltip = {
			text = scandinavian_nationalism_progress_at_100p_tt
			has_global_variable = scandinavian_nationalism_progress
			global_var:scandinavian_nationalism_progress >= 1000
		}
	}
	on_weekly_pulse = {
		effect = {
			# For the tooltip, find the difference between current progress and previous weeks progress
			save_scope_value_as = {
				name = old_scandinavian_nationalism_progress
				value = global_var:scandinavian_nationalism_progress
			}
			## WEEKLY CALCULATIONS FOR SCANDINAVIAN NATIONALISM PROGRESS BAR ##
			# If the current country has okay relations with another Scandi country...
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root >= relations_threshold:cordial
					}
				}
				change_global_variable = {
					name = scandinavian_nationalism_progress
					add = {
						# Add to the progress bar, depending on how positive the relations are
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:cordial
							}
							add = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:amicable
							}
							add = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root >= relations_threshold:friendly
							}
							add = scandinavian_nationalism_progress_tick_small
						}
					}
				}
			}
			# And if the current country has poor relations with another Scandi country...
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root <= relations_threshold:poor
					}
				}
				change_global_variable = {
					name = scandinavian_nationalism_progress
					add = {
						# This is gonna be a negative given the below conditions, so its fine to use add here
						# Subtract from the progress bar, depending on how negative the relations are
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:poor
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:cold
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
						every_country = {
							limit = {
								has_journal_entry = je_scandinavian_nationalism
								relations:root <= relations_threshold:hostile
							}
							subtract = scandinavian_nationalism_progress_tick_small
						}
					}
				}
			}
			# If at war with fellow Scandinavian
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_war_with = root
					}
				}
				every_country = {
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						has_war_with = root
					}
					change_global_variable = {
						name = scandinavian_nationalism_progress
						subtract = scandinavian_nationalism_progress_tick_large
					}
				}
			}
			# If in a war against non-Scandi country, with an independent Scandi as your ally
			if = {
				limit = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_ally_of = root
					}
					NOT = {
						any_country = {
							has_journal_entry = je_scandinavian_nationalism
							is_diplomatic_play_enemy_of = root
						}
					}
				}
				# This is a bit wonky.. If there's a diplo play against a non-Scandi featuring 2 Scandis, and an inter-Scandi war going on it'll cancel each other out
				every_country = {
					limit = {
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_ally_of = root
					}
					change_global_variable = {
						name = scandinavian_nationalism_progress
						add = scandinavian_nationalism_progress_tick_medium
					}
				}
			}
			## END OF WEEKLY CALCULATIONS
			# Clamp that variable so it stays between 0 and 1000
			clamp_global_variable = {
				name = scandinavian_nationalism_progress
				min = 0
				max = 1000
			}
			# For the tooltip, find the difference between current progress and previous weeks progress
			set_global_variable = {
				name = scandinavian_nationalism_progress_change_since_last_week
				value = {
					value = global_var:scandinavian_nationalism_progress
					subtract = scope:old_scandinavian_nationalism_progress
				}
			}
		}
	}
	# A special status description for the Scandinavian Nationalism progress bar
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = status_scandinavian_nationalism
				trigger = {
					has_global_variable = scandinavian_nationalism_progress
				}
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_scandinavian_nationalism_positive_relations
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root >= relations_threshold:cordial
					}
				}
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_scandinavian_nationalism_negative_relations
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						relations:root <= relations_threshold:poor
					}
				}
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_scandinavian_nationalism_infighting
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						has_war_with = root
					}
				}
			}
		}
	}
	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_scandinavian_nationalism_fighting_against_outsiders
				trigger = {
					any_country = {
						has_journal_entry = je_scandinavian_nationalism
						is_diplomatic_play_ally_of = root
					}
					NOT = {
						any_country = {
							has_journal_entry = je_scandinavian_nationalism
							is_diplomatic_play_enemy_of = root
						}
					}
				}
			}
		}
	}
	weight = 100
	should_be_pinned_by_default = yes
}

			any_primary_culture = {
				has_discrimination_trait = heritage_nordic
			}
		}

		custom_tooltip = {
			text = scandinavian_nationalism_progress_at_25p_tt

	}
	complete = {

			filter = {

			count = all

			has_technology_researched = pan-nationalism
		}
	}

		add_modifier = {
			name = scandinavian_currency_union
		}
	}
}

		OR = {
			has_journal_entry = je_scandinavian_freedom_of_movement_act
		}
		OR = {
			any_country = {
				count = 1
				any_primary_culture = {
					has_discrimination_trait = heritage_nordic
				}
				is_subject = no
				any_subject_or_below = {
					any_primary_culture = {
						has_discrimination_trait = heritage_nordic
					}
				}
			}
			any_country = {
				filter = {
					has_journal_entry = je_scandinavian_freedom_of_movement_act
				}
				count = all
				OR = {
					has_law = law_type:law_no_migration_controls
					this.market = ROOT.market
				}
			}
		}
	}
	on_complete = {
		add_modifier = {
			name = scandinavian_freedom_of_movement_act
		}
	}
