# 1st Hungarian Revolution, happens after Frankfurt ends
je_imperia_hungarian_revolution = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content
	scripted_button = imperia_je_imperia_hungarian_revolution_ultimatum_button
	is_shown_when_inactive = {
		OR = {
			has_global_variable = peoples_springtime_happening
			has_global_variable = peoples_springtime_happened
		}
		exists = c:HUN
		c:HUN = root
		is_subject = yes
	}
	possible = {
		custom_tooltip = {
			text = frankfurt_parliament_is_over_tt
			imperia_frankfurt_parliament_is_in_session = no
			has_global_variable = frankfurt_parliament_held
		}
	}
	immediate = {
		set_global_variable = je_imperia_hungarian_revolution_was_here
	}
	fail = {
		custom_tooltip = {
			text = lost_hungarian_revolution_tt
			has_variable = lost_hungarian_revolution
		}
	}
	on_fail = {
		every_scope_character = {
			limit = {
				has_template = agitator_lajos_kossuth
			}
			exile_character = yes
		}
	}
	complete = {
		is_subject = no
		is_at_war = no
	}
	weight = 5000
	should_be_pinned_by_default = yes
}
