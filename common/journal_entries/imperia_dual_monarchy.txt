# 1st Hungarian Revolution, happens after Frankfurt ends
je_imperia_hungarian_revolution = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	group = je_group_historical_content
	scripted_button = imperia_je_imperia_hungarian_revolution_ultimatum_button
	is_shown_when_inactive = {
		OR = {
			has_global_variable = peoples_springtime_happening
			has_global_variable = peoples_springtime_happened
		}
		exists = c:HUN
		c:HUN = root
		is_subject = yes
	}
	possible = {
		custom_tooltip = {
			text = frankfurt_parliament_is_over_tt
			imperia_frankfurt_parliament_is_in_session = no
			has_global_variable = frankfurt_parliament_held
		}
	}
	fail = {
		custom_tooltip = {
			text = lost_hungarian_revolution_tt
			has_variable = lost_hungarian_revolution
		}
	}
	on_fail = {
		every_scope_character = {
			limit = {
				has_template = agitator_lajos_kossuth
			}
			exile_character = yes
		}
	}
	complete = {
		is_subject = no
		is_at_war = no
	}
	weight = 5000
	should_be_pinned_by_default = yes
}

# Austrian neo absolutism
# Identical to the Hungarian passive resistance, just inverse
je_imperia_austrian_neo_absolutism = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	group = je_group_historical_content
	scripted_button = imperia_austria_punish_extra_land_button
	scripted_button = imperia_austria_punish_transylvania_button
	scripted_button = imperia_austria_punish_vojvodina_button
	scripted_button = imperia_austria_punish_croatia_button
	scripted_button = imperia_austria_punish_slovakia_button
	scripted_progress_bar = imperia_passive_resistance_suppression_progress_bar
	scripted_progress_bar = imperia_passive_resistance_hungarian_resistance_progress_bar
	scripted_progress_bar = imperia_passive_resistance_deadlock_progress_bar
	modifiers_while_active = {
		je_imperia_austrian_neo_absolutism_modifier
	}
	fail = {
		any_diplomatic_play = {
			count = 0
			is_diplomatic_play_type = dp_independence
			initiator_is = c:HUN
		}
		OR = {
			imperia_passive_resistance_hungarian_resistance_won_trigger = yes
			imperia_passive_resistance_deadlock_won_trigger = yes
		}
	}
	on_fail = {
		show_as_tooltip = {
			custom_label = if_je_completed_from_hungarian_resistance_tt
			imperia_passive_resistance_hungarian_resistance_won_effect = yes
		}
		show_as_tooltip = {
			custom_label = if_je_completed_from_deadlock_tt
			imperia_passive_resistance_deadlock_won_effect = yes
		}
		hidden_effect = {
			if = {
				limit = {
					imperia_passive_resistance_hungarian_resistance_won_trigger = yes
				}
				imperia_passive_resistance_hungarian_resistance_won_effect = yes
			}
			if = {
				limit = {
					imperia_passive_resistance_deadlock_won_trigger = yes
				}
				imperia_passive_resistance_deadlock_won_effect = yes
			}
			every_scope_state = {
				limit = {
					has_modifier = imperia_passive_resistance_state_modifier
				}
				remove_modifier = imperia_passive_resistance_state_modifier
			}
		}
	}
	complete = {
		any_diplomatic_play = {
			count = 0
			is_diplomatic_play_type = dp_independence
			initiator_is = c:HUN
		}
		imperia_passive_resistance_suppression_won_trigger = yes
	}
	on_complete = {
		imperia_passive_resistance_suppression_won_effect = yes
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_modifier = imperia_passive_resistance_state_modifier
				}
				remove_modifier = imperia_passive_resistance_state_modifier
			}
		}
	}
	invalid = {
		c:HUN = {
			is_subject = no
		}
	}
	on_invalid = {
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_modifier = imperia_passive_resistance_state_modifier
				}
				remove_modifier = imperia_passive_resistance_state_modifier
			}
		}
		remove_global_variable ?= passive_resistance_deadlock_difference
	}
	weight = 5000
	should_be_pinned_by_default = yes
}

# Hungarian passive resistance
# Identical to the Austrian neo absolutism, just inverse
je_imperia_passive_resistance = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	group = je_group_historical_content
	scripted_button = imperia_passive_resistance_commission_art_button
	scripted_button = imperia_passive_resistance_underground_network_button
	scripted_button = imperia_passive_resistance_publish_essay_button
	scripted_button = imperia_passive_resistance_petition_austria_button
	scripted_progress_bar = imperia_passive_resistance_suppression_progress_bar
	scripted_progress_bar = imperia_passive_resistance_hungarian_resistance_progress_bar
	scripted_progress_bar = imperia_passive_resistance_deadlock_progress_bar
	modifiers_while_active = {
		je_imperia_passive_resistance_modifier
	}
	immediate = {
		scope:journal_entry = {
			set_global_variable = {
				name = passive_resistance_deadlock_difference
				value = {
					value = 0
					add = "scripted_bar_progress(imperia_passive_resistance_hungarian_resistance_progress_bar)"
					subtract = "scripted_bar_progress(imperia_passive_resistance_suppression_progress_bar)"
				}
			}
		}
		set_variable = {
			name = passive_resistance_underground_network_strength
			value = 1
		}
		# Set the art type to be commissioned...
		random_list = {
			# Set next to be a poem
			10 = {
				set_variable = {
					name = hungarian_art_type
					value = 1
				}
			}
			# Set next to be a play
			10 = {
				set_variable = {
					name = hungarian_art_type
					value = 2
				}
			}
			# Set next to be a piece of music
			10 = {
				set_variable = {
					name = hungarian_art_type
					value = 3
				}
			}
		}
	}
	fail = {
		any_diplomatic_play = {
			count = 0
			is_diplomatic_play_type = dp_independence
			initiator_is = c:HUN
		}
		OR = {
			imperia_passive_resistance_suppression_won_trigger = yes
			imperia_passive_resistance_deadlock_won_trigger = yes
		}
	}
	on_fail = {
		show_as_tooltip = {
			custom_label = if_je_completed_from_suppression_tt
			imperia_passive_resistance_suppression_won_effect = yes
		}
		show_as_tooltip = {
			custom_label = if_je_completed_from_deadlock_tt
			imperia_passive_resistance_deadlock_won_effect = yes
		}
		hidden_effect = {
			if = {
				limit = {
					imperia_passive_resistance_suppression_won_trigger = yes
				}
				imperia_passive_resistance_suppression_won_effect = yes
			}
			if = {
				limit = {
					imperia_passive_resistance_deadlock_won_trigger = yes
				}
				imperia_passive_resistance_deadlock_won_effect = yes
			}
		}
	}
	complete = {
		any_diplomatic_play = {
			count = 0
			is_diplomatic_play_type = dp_independence
			initiator_is = c:HUN
		}
		imperia_passive_resistance_hungarian_resistance_won_trigger = yes
	}
	on_complete = {
		imperia_passive_resistance_hungarian_resistance_won_effect = yes
	}
	invalid = {
		is_subject = no
	}
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				set_global_variable = {
					name = passive_resistance_deadlock_difference
					value = {
						value = 0
						add = "scripted_bar_progress(imperia_passive_resistance_hungarian_resistance_progress_bar)"
						subtract = "scripted_bar_progress(imperia_passive_resistance_suppression_progress_bar)"
					}
				}
			}
		}
	}
	on_monthly_pulse = {
		effect = {
			# Removes the passive resistance modifier from states Austria confiscated if they get passed back to Hungary on a monthly basis
			if = {
				limit = {
					any_scope_state = {
						has_modifier = imperia_passive_resistance_state_modifier
					}
				}
				every_scope_state = {
					has_modifier ?= imperia_passive_resistance_state_modifier
				}
			}
		}
	}
	weight = 5000
	should_be_pinned_by_default = yes
}
