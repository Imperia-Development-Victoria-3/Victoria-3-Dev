crownland_construction_penalty_value = {
	value = modifier:country_construction_add
	# If the country has a variable for the penalty, add it
	# Here so it doesnt start flip flopping like we saw with currency transfer
	if = {
		limit = {
			has_variable = crownland_construction_penalty_value_variable
		}
		add = var:crownland_construction_penalty_value_variable
	}
	multiply = imperia_crownland_construction_transfer_percentage
}

crownland_construction_bonus_value = {
	value = 0
	every_direct_subject = {
		limit = {
			is_subject_type = subject_type_crownland
		}
		add = var:crownland_construction_penalty_value_variable
	}
}
