on_imperia_monthly_pulse_timegated_military_index = {
	on_actions = {
        # Army
        trench_infantry_timegate_check
        squad_infantry_timegate_check
        mechanized_infantry_timegate_check
        shrapnel_artillery_timegate_check
        siege_artillery_timegate_check
        # Navy
        monitors_timegate_check
        destroyers_timegate_check
        scout_cruisers_timegate_check
        ironclads_timegate_check
        dreadnoughts_timegate_check
        battleships_timegate_check
        torpedo_boats_timegate_check
        submarines_timegate_check
        carriers_timegate_check
	}
}

trench_infantry_timegate_check = {
	trigger = {
		year >= trench_infantry_year
		has_technology_researched = trench_works
		NOT = {
			has_technology_researched = trench_infantry_timegate_technology
		}
	}
	effect = {
		add_technology_researched = trench_infantry_timegate_technology
	}
}
