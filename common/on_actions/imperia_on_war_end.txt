# Root = Diplomatic Play
# scope:actor = Initiator country
# scope:target = Target country
on_imperia_war_end = {
	effect = {
		# If this is a Hungarian revolt against Austria, remove the foreign aid modifiers from everyone
		if = {
			limit = {
				scope:actor = c:HUN
				scope:target = c:AUS
				is_diplomatic_play_type = dp_independence
			}
			every_country = {
				# Ideally we'd check if the country got the event, but... Oh well
				# Luckily doesnt do anything if the country doesnt have any of the modifiers
				imperia_hungarian_revolution_remove_aid_effect = yes
			}
            # Furthermore, if this is the first time the Hungarian revolt is lost, exile all agitators and set the lost revolution variable
            if = {
                limit = {
                    c:HUN = {
                        NOT = {
                            has_variable = lost_hungarian_revolution
                        }
                        NOT = {
                            is_diplomatic_play_enemy_of = c:AUS
                        }
                    }
                    has_global_variable = hungarian_revolution_started
                }
                c:HUN = {
                    set_variable = lost_hungarian_revolution
                    # Remember to get all the laws
                    copy_laws = overlord
                }
                if = {
                    limit = {
                        c:HUN.overlord = c:AUS
                    }
                    c:AUS = {
                        trigger_event = {
                            id = imperia_dual_monarchy.3
                            popup = yes
                        }
                        add_journal_entry = {
                            type = je_imperia_austrian_neo_absolutism
                        }
                        # Just to make sure Austria doesnt make Hungary a dominion or something
                        create_diplomatic_pact = {
                            country = c:HUN
                            type = crownland
                        }
                    }
                    c:HUN = {
                        add_journal_entry = {
                            type = je_imperia_passive_resistance
                        }
                    }
                }
            }
		}
	}
}
