# root = law
# type = law_type
# owner = the country activating the law
# Courtesy of VTMs headaches
on_imperia_law_activated = {
	effect = {
		if = {
			limit = {
				owner ?= c:GBR
				OR = {
					type = law_type:law_parliamentary_republic
					type = law_type:law_presidential_republic
				}
			}
			owner = {
				set_variable = can_change_british_flag
			}
		}
		
		if = {
			limit = {
				OR = {
					type = law_type:law_traditionalism
					type = law_type:law_interventionism
					type = law_type:law_agrarianism
					type = law_type:law_industry_banned
					type = law_type:law_extraction_economy
					type = law_type:law_laissez_faire
					type = law_type:law_cooperative_ownership
					type = law_type:law_command_economy
				}
			}
			owner = {
				imperia_auto_apply_private_allocation = yes
			}
		}
	}
}
